def converter(lakh):
    return pd.Series({"Lakh INR": lakh, "INR": lakh*LAKH,
                      "US$": lakh*LAKH/INR_POR_USD, "R$": lakh*LAKH*BRL_POR_INR})

def n_outliers(s):
    q1, q3 = s.quantile([.25, .75]); i = q3 - q1

# BLOCO 7 - Separacao treino/teste e pre-processamento
# ==============================================================
CAT = ["Fuel_Type", "Seller_Type", "Transmission"]
NUM = ["Present_Price", "Kms_Driven", "Owner", "Car_Age"]

X = carros[NUM + CAT]
y = carros["Selling_Price"]
X_tr, X_te, y_tr, y_te = train_test_split(X, y, test_size=0.2, random_state=SEED)
print("Treino:", X_tr.shape, "| Teste:", X_te.shape)

def criar_pipeline(modelo, escalar):
    """One-Hot nas categoricas; StandardScaler so quando o modelo e sensivel a escala."""
    pre = ColumnTransformer([
        ("cat", OneHotEncoder(drop="first", handle_unknown="ignore"), CAT),
        ("num", StandardScaler() if escalar else "passthrough", NUM),
    ])
    return Pipeline([("pre", pre), ("m", modelo)])

# BLOCO 8 - Treinamento e ajuste de hiperparametros
# GridSearchCV com 5 folds, usando APENAS o conjunto de treino
# ==============================================================
t0 = time.time()
configs = {
    "Regressao Linear Multipla": (criar_pipeline(LinearRegression(), True), {}),
    "Random Forest": (criar_pipeline(RandomForestRegressor(random_state=SEED), False),
                      {"m__n_estimators": [200, 500], "m__max_depth": [None, 5, 10],
                       "m__min_samples_leaf": [1, 2, 4]}),
    "SVR": (criar_pipeline(SVR(), True),
            {"m__kernel": ["rbf", "linear"], "m__C": [1, 10, 100, 300],
             "m__epsilon": [0.05, 0.1, 0.5], "m__gamma": ["scale", 0.01, 0.1]}),
}

melhores = {}
for nome, (pipe, grade) in configs.items():
    if grade:
        gs = GridSearchCV(pipe, grade, cv=5, scoring="neg_mean_absolute_error", n_jobs=1)
        gs.fit(X_tr, y_tr)
        melhores[nome] = gs.best_estimator_
        print(f"{nome}: {gs.best_params_}   ({time.time()-t0:.0f}s)", flush=True)
    else:
        melhores[nome] = pipe.fit(X_tr, y_tr)
        print(f"{nome}: sem hiperparametros a ajustar", flush=True)
print("\nModelos treinados com sucesso!")

# ==============================================================
# BLOCO 9 - Metricas: teste, treino (overfitting) e validacao cruzada
# ==============================================================
cv = RepeatedKFold(n_splits=5, n_repeats=3, random_state=SEED)
linhas, preds = [], {}
for nome, mod in melhores.items():
    p_te, p_tr = mod.predict(X_te), mod.predict(X_tr)
    r = cross_validate(mod, X, y, cv=cv, scoring=("r2", "neg_mean_absolute_error"))
    preds[nome] = p_te
    linhas.append({
        "Modelo": nome,
        "R2 teste": r2_score(y_te, p_te),
        "MAE teste": mean_absolute_error(y_te, p_te),
        "MAPE teste (%)": mean_absolute_percentage_error(y_te, p_te) * 100,
        "RMSE teste": np.sqrt(mean_squared_error(y_te, p_te)),
        "R2 treino": r2_score(y_tr, p_tr),
        "R2 CV (media)": r["test_r2"].mean(),
        "R2 CV (desvio)": r["test_r2"].std(),
        "MAE CV": -r["test_neg_mean_absolute_error"].mean(),
    })
tabela = pd.DataFrame(linhas).set_index("Modelo")
tabela["Gap R2 (treino - teste)"] = tabela["R2 treino"] - tabela["R2 teste"]
print("=== TABELA COMPARATIVA DE METRICAS (valores em lakh INR) ===")
tabela.round(3)
